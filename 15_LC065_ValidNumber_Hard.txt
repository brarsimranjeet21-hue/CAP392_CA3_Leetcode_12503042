class Solution {
public:
    bool isNumber(string s) {

        bool digit = false;
        bool dot = false;
        bool e = false;

        for (int i = 0; i < s.length(); i++) {

            char ch = s[i];

            if (ch >= '0' && ch <= '9') {
                digit = true;
            }

            else if (ch == '+' || ch == '-') {
                if (i != 0 && s[i - 1] != 'e' && s[i - 1] != 'E') {
                    return false;
                }
            }

            else if (ch == '.') {
                if (dot || e) {
                    return false;
                }
                dot = true;
            }

            else if (ch == 'e' || ch == 'E') {
                if (e || !digit) {
                    return false;
                }
                e = true;
                digit = false;
            }

            else {
                return false;
            }
        }

        return digit;
    }
};
