class Solution {
public:
    vector<string> fullJustify(vector<string>& words, int maxWidth) {

        vector<string> ans;
        int n = words.size();
        int i = 0;

        while (i < n) {

            int j = i;
            int letters = 0;

            // Find how many words can fit in one line
            while (j < n && letters + words[j].length() + (j - i) <= maxWidth) {
                letters += words[j].length();
                j++;
            }

            int spaces = maxWidth - letters;
            int gaps = j - i - 1;

            string line = "";

            // Last line or only one word
            if (j == n || gaps == 0) {

                for (int k = i; k < j; k++) {
                    line += words[k];

                    if (k < j - 1) {
                        line += " ";
                    }
                }

                while (line.length() < maxWidth) {
                    line += " ";
                }
            }

            else {

                int spaceEach = spaces / gaps;
                int extra = spaces % gaps;

                for (int k = i; k < j; k++) {

                    line += words[k];

                    if (k < j - 1) {

                        int count = spaceEach;

                        if (extra > 0) {
                            count++;
                            extra--;
                        }

                        for (int x = 0; x < count; x++) {
                            line += " ";
                        }
                    }
                }
            }

            ans.push_back(line);
            i = j;
        }

        return ans;
    }
};
